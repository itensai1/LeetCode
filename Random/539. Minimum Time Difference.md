public class Solution
{ 
    static int toMinutes(string timePoint) => Convert.ToInt32(timePoint.Substring(0,2)) * 60 + Convert.ToInt32(timePoint.Substring(3,2));

    public int FindMinDifference(IList<string> timePoints)
    {
        List<int> minutes = new List<int>();
        for (int i = 0; i < timePoints.Count; i++) minutes.Add(toMinutes(timePoints[i])) ;
        minutes.Sort();

        int answer = toMinutes("24:00") - minutes.Last() + minutes[0];

        for (int i = 1; i < minutes.Count; i++){
            answer = Math.Min(answer, minutes[i] - minutes[i - 1]);
        }
                
        return answer;
    }
}

class Solution {
public:
    static int toMinutes(string timePoint){return stoi(timePoint.substr(0,2)) * 60 + stoi(timePoint.substr(3,2));}

    int findMinDifference(vector<string>& timePoints) {
        vector<int>minutes;
        for (string s:timePoints) minutes.push_back(toMinutes(s));

        sort(minutes.begin(), minutes.end());

        int answer = toMinutes("24:00") - minutes.back() + minutes[0];

        for (int i = 1; i < minutes.size() ; ++i) {
            answer = min(answer, minutes[i] - minutes[i-1]);
        }

        return answer;
    }
};

class Solution:
    def findMinDifference(self, timePoints: List[str]) -> int:

        def to_minutes(timePoint :str) -> int:
            return int(timePoint[:2])*60 + int(timePoint[3:])

        minutes:List[int] = sorted(list(map(to_minutes,timePoints)))

        answer:int = to_minutes("24:00") - minutes[-1] + minutes[0]

        for i in range(1,len(minutes)):
            answer = min(answer, minutes[i] - minutes[i-1])
        
        return answer
